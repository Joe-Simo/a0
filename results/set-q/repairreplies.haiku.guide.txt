fn post (u32,bool) u32 u32x4 -> (u32,bool)
v get p2 p1
z at p0 0
a add z v
s sub z v
m rem p1 2
e eq m 0
n select e a s
neg lt n 0
o at p0 1
f or o neg
r put p0 1 f
final put r 0 n
ret final
end
