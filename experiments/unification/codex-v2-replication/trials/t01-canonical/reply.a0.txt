fn lexstep u32 u32 u32 u32 u32 -> (bool,u32,u32,u32,u32,u32,u32)
c mov p2

k5 eq p0 5
k7 eq p0 7
k9 eq p0 9
k10 eq p0 10
inw or k1 k2
notw select isw false true
wclose and inw notw
escd eq p4 1
notesc select escd false true
sq and k3 isq
sclose and sq notesc
cclose and k8 isnl
s57 or k5 k7

arrow and arrowstate isgt

emit or e1 arrowstate
c0 or e0 cclose

consumed or sclose arrow
ek0 select k6 6 p0

ek1 select arrow 4 ekbase
ek select sclose 3 ek1

free and none notcons

n0 select issp 0 9
n1 select isat 7 n0
n2 select isminus 6 n1

n3 select isnl 5 n2e
n4 select ishash 8 n3
n5 select isq 3 n4
n6 select isw wk n5

nk select free n6 cont
ns select free p3 p1
e2 select isbs 1 0
e3 select escd 0 e2
esc select k3 e3 0
r rec emit ek es el nk ns esc
ret r
end

fn lexsrc u32x512x64 u32 -> (u32x128x384,u32)
zero call ztok
n127 add p1 127
np shr n127 7
s0 rec zero 0 0 0 0
s fold lexpage np s0 p0 p1
out at s 0
nout at s 1
pk at s 2
ps at s 3
k1 eq pk 1
k2 eq pk 2
k5 eq pk 5
k6 eq pk 6
k7 eq pk 7
k9 eq pk 9
k10 eq pk 10
w or k1 k2
o0 or k5 k6
o1 or o0 k7
o2 or o1 k9
one or o2 k10
pend or w one
wlen sub p1 ps
len select w wlen 1
ek select k10 9 pk
c call tokput3 out nout ek ps len pend
n3 add nout 3
nf select pend n3 nout
r rec c nf
ret r
end
