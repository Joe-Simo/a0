fn countbit u32 u32 u32 -> u32
bit and shr C B 1
total add A bit

fn countrow u32 u32 u32x32 -> u32
row get C B
bits fold countbit 32 0 row
total add A bits

fn writegrid u32x32 io -> io
population fold countrow 32 0 A
header write B 76
length write header 32
rows fold writerow 32 length A
t write rows population
