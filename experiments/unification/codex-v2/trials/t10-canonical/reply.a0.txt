fn r_int io u32 u32 -> io
q div p2 100
remnant rem p2 100
up ge remnant 50
inc select up 1 0
rounded add q inc
r call putnum p0 rounded
ret r
end

fn r_one io u32 u32 -> io
tn div p2 10
remnant rem p2 10
up ge remnant 5
inc select up 1 0
rounded add tn inc
q div rounded 10
a call putnum p0 q
b write a 2
dot text "."
c puts b dot
td rem rounded 10
ch add td 48
d write c 2
e write d 1
f write e ch
ret f
end
