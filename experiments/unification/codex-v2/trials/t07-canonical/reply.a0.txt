fn wire_reverse_step u32 u32 u32 -> u32
shifted shl p0 1
source shr p2 p1
bit and source 1
out or shifted bit
ret out
end

fn wire_reverse32 u32 -> u32
out fold wire_reverse_step 32 0 p0
ret out
end

fn readrow u32x32 u32 io -> (u32x32,io)
r read p2
v at r 0
t at r 1
converted call wire_reverse32 v
g set p0 p1 converted
out rec g t
ret out
end
