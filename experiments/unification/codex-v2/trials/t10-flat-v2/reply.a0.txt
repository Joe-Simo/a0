fn r_int io _ _ -> io
q div C 100
remn rem C 100
up ge remn 50
inc select up 1 0
rounded add q inc
r putnum A rounded
ret r
end

fn r_one io _ _ -> io
tn div C 10
remn rem C 10
up ge remn 5
inc select up 1 0
rounded add tn inc
q div rounded 10
a putnum A q
b write a 2
dot "."
c puts b dot
td rem rounded 10
ch add td 48
d write c 2
e write d 1
f write e ch
ret f
end
